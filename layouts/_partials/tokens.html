{{- $html := string . -}} {{- $names := dict "⌘" "Command" "⌥" "Option" "⇧" "Shift" "⌃" "Control" "↩" "Return" "⌫" "Delete" "⎋" "Escape" "↑" "Up Arrow" "↓" "Down Arrow" "←" "Left Arrow" "→" "Right Arrow" "⇥" "Tab" -}} {{- range findRESubmatch `\[\[([^\]]+)\]\]` $html -}} {{- $token := index . 0 -}} {{- $inner := index . 1 -}} {{- $replacement := "" -}} {{- if strings.HasPrefix $inner "sf:" -}} {{- $name := strings.TrimPrefix "sf:" $inner -}} {{- with index (resources.Match (printf "symbols/%s_*.png" (replace $name "." "-"))) 0 -}} {{- $replacement = printf `` .RelPermalink (div .Width 4) (div .Height 4) -}} {{- else -}} {{- errorf "symbol %s is not rendered into assets/symbols" $name -}} {{- end -}} {{- else -}} {{- $keys := slice -}} {{- $parts := split $inner " " -}} {{- $last := sub (len $parts) 1 -}} {{- range $i, $part := $parts -}} {{- $name := index $names $part -}} {{- if and (eq $part "/") (gt $i 0) (lt $i $last) -}} {{- $keys = $keys | append `/or` -}} {{- else if $name -}} {{- $keys = $keys | append (printf `%s%s` $part $name) -}} {{- else if eq $part "space" -}} {{- $keys = $keys | append `Space` -}} {{- else -}} {{- $keys = $keys | append (printf `%s` (upper $part)) -}} {{- end -}} {{- end -}} {{- $replacement = printf `%s` (delimit $keys "") -}} {{- end -}} {{- $html = replace $html $token $replacement -}} {{- end -}} {{- return ($html | safeHTML) -}}
